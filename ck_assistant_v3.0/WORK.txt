   def get_selected_lorebooks(session_name):
       cfg = load_config()
       return cfg.get("session_lorebooks", {}).get(session_name, []) or []

   def set_selected_lorebooks(session_name, book_names):
       cfg = load_config()
       cfg.setdefault("session_lorebooks", {})
       cfg["session_lorebooks"][session_name] = list(book_names or [])
       save_config(cfg)
       return True

   def create_lorebook(name, description="", terms=None):
       cfg = load_config()
       cfg.setdefault("lorebooks", {})
       if name in cfg["lorebooks"]:
           return False
       cfg["lorebooks"][name] = {
           "description": description,
           "terms": terms or {},
           "last_updated": datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
       }
       save_config(cfg)
       return True

六、旧数据迁移（load_config 内自动）
--------------------------------------------
if "selected_lorebooks" in config and config["selected_lorebooks"]:
    cfg.setdefault("session_lorebooks", {})
    cur = config.get("current_session", "默认单聊")
    current_picks = config["session_lorebooks"].get(cur, [])
    for b in config["selected_lorebooks"]:
        if b not in current_picks:
            current_picks.append(b)
    config["session_lorebooks"][cur] = current_picks
    config["selected_lorebooks"] = []
    save_config(config)
